# =====================================================================
# Ghi nhật ký
# =====================================================================

function Write-ITLog {

    param(
        [string]$Status,
        [string]$Message
    )

    try {

        $Time = Get-Date -Format "dd/MM/yyyy HH:mm:ss"

        "$Time | $Status | $Message" |
            Out-File `
                -FilePath $LogFile `
                -Append `
                -Encoding UTF8
    }
    catch {
    }
}


# =====================================================================
# Hiển thị thông báo cho người dùng
#
# Popup:
# - Không dấu
# - Chữ in hoa
#
# MSG.EXE được gọi thông qua CMD.EXE vì phương án này đã hoạt động
# ổn định trong quá trình kiểm tra thực tế.
# =====================================================================

function Show-UserMessage {

    param(
        [string]$Message
    )


    $MessageFile = Join-Path `
        $env:TEMP `
        "IT-MAINTENANCE-MESSAGE-$PID.txt"


    try {

        # Lưu nội dung thông báo vào file tạm
        $Message |
            Set-Content `
                -Path $MessageFile `
                -Encoding Unicode


        # Chuyển nội dung file vào MSG.EXE
        $Command = "type `"$MessageFile`" | `"$env:SystemRoot\System32\msg.exe`" * /TIME:300"


        # Chạy CMD riêng để gửi thông báo
        $Process = Start-Process `
            -FilePath "$env:SystemRoot\System32\cmd.exe" `
            -ArgumentList "/d", "/c", $Command `
            -WindowStyle Hidden `
            -Wait `
            -PassThru


        if ($Process.ExitCode -eq 0) {

            Write-ITLog `
                "THONG BAO" `
                "MSG.EXE DA GUI THONG BAO THANH CONG. EXITCODE=0."

            return $true
        }
        else {

            Write-ITLog `
                "CANH BAO" `
                "MSG.EXE TRA VE EXITCODE=$($Process.ExitCode)."

            return $false
        }
    }
    catch {

        Write-ITLog `
            "LOI THONG BAO" `
            "KHONG GUI DUOC THONG BAO. $($_.Exception.Message)"

        return $false
    }
    finally {

        Remove-Item `
            -Path $MessageFile `
            -Force `
            -ErrorAction SilentlyContinue
    }
}


# =====================================================================
# Chạy DiskPart
# =====================================================================

function Invoke-ITDiskPart {

    param(
        [string[]]$Commands
    )


    $TempFile = Join-Path `
        $env:TEMP `
        "IT-MAINTENANCE-DISKPART-$PID.txt"


    try {

        $Commands |
            Set-Content `
                -Path $TempFile `
                -Encoding ASCII


        $Output = & "$env:SystemRoot\System32\diskpart.exe" `
            /s `
            $TempFile `
            2>&1


        return ($Output -join " | ")
    }
    finally {

        Remove-Item `
            -Path $TempFile `
            -Force `
            -ErrorAction SilentlyContinue
    }
}


# =====================================================================
# Tìm Volume theo GUID
# =====================================================================

function Get-ITVolume {

    param(
        [string]$GUID
    )


    return (
        Get-CimInstance Win32_Volume |
            Where-Object {
                $_.DeviceID -eq $GUID
            } |
            Select-Object -First 1
    )
}


# =====================================================================
# Đọc thông tin dung lượng Volume
# =====================================================================

function Get-ITDiskInfo {

    param(
        $Volume
    )


    if ($null -eq $Volume) {

        return $null
    }


    if ($Volume.Capacity -le 0) {

        return $null
    }


    $TotalGB = [math]::Round(
        $Volume.Capacity / 1GB,
        2
    )


    $FreeGB = [math]::Round(
        $Volume.FreeSpace / 1GB,
        2
    )


    $UsedGB = [math]::Round(
        ($Volume.Capacity - $Volume.FreeSpace) / 1GB,
        2
    )


    return [PSCustomObject]@{

        TotalGB     = $TotalGB
        UsedGB      = $UsedGB
        FreeGB      = $FreeGB
    }
}

                        $DefragText = ($DefragResult | ForEach-Object { $_.ToString() }) -join " | "

                            $HasDiskError = $true
                            if ($ErrorDrives -notcontains $Drive) { $ErrorDrives += $Drive }
                        }
                    }
                    catch {
                        Write-ITLog "LOI DEFRAG" "$Drive | $($_.Exception.Message)"

                    }
                }

                if ($ErrorDrives -notcontains $Drive) { $ErrorDrives += $Drive }
            }
        }

THONG BAO IT-MAINTENANCE

MAY TINH: $ComputerName

PHAT HIEN VAN DE TREN O DIA: $ErrorDriveText

VUI LONG LIEN HE QUAN TRI VIEN

QUAN TRI VIEN
"@

DA SU DUNG      : $UsedGB GB

VUI LONG DON RAC.

VUI LONG KIEM TRA, TINH LOC VA XOA
CAC DU LIEU KHONG CON CAN THIET.

KIEM TRA THU MUC TAM, THUNG RAC
VA CAC DU LIEU TRUNG LAP NEU CO.

CAC DU LIEU CON CAN THIET CAN DUOC
SAP XEP HOAC DI CHUYEN PHU HOP.

NEU DUNG LUONG TRONG GIAM XUONG <20 GB,
HE THONG SE THONG BAO VA CHO 60 PHUT
DE NGUOI DUNG XU LY.

NEU SAU 60 PHUT DUNG LUONG VAN <20 GB,
HE THONG SE TAM THOI AN O D:.

DU LIEU HIEN CO KHONG BI XOA.

QUAN TRI VIEN
"@


        $MessageSent = Show-UserMessage `
            -Message $Message


        if ($MessageSent) {

            Write-ITLog `
                "THONG BAO" `
                "DA GUI CANH BAO O D: SAP DAY."
        }
        else {

            Write-ITLog `
                "CANH BAO" `
                "LENH GUI CANH BAO O D: SAP DAY KHONG THANH CONG."
        }


        exit 0
    }


    # =================================================================
    # D. Dung lượng trống <20 GB
    # =================================================================

    if ($FreeGB -lt $HideGB) {


        Write-ITLog `
            "NGHIEM TRONG" `
            "O D: SAP DAY. DUNG LUONG TRONG CHI CON $FreeGB GB."


        # -------------------------------------------------------------
        # Thông báo cho người dùng và bắt đầu thời gian xử lý 30 phút
        # -------------------------------------------------------------

        $Message = @"
THONG BAO IT-MAINTENANCE

O DIA D: SAP DAY!

TONG DUNG LUONG : $TotalGB GB
DA SU DUNG      : $UsedGB GB
CON TRONG       : $FreeGB GB

VUI LONG DON RAC NGAY.

DUNG LUONG TRONG CUA O D:
DA XUONG <20 GB.

BAN CO 60 PHUT DE KIEM TRA
VA DON DEP DU LIEU KHONG CAN THIET.

HE THONG SE KIEM TRA LAI SAU 60 PHUT.

NEU DUNG LUONG TRONG VAN <20 GB,
O D: SE TAM THOI BI AN.

DU LIEU HIEN CO KHONG BI XOA.

QUAN TRI VIEN
"@


        $MessageSent = Show-UserMessage `
            -Message $Message


        if ($MessageSent) {

            Write-ITLog `
                "THONG BAO" `
                "DA GUI CANH BAO DUNG LUONG O D: <20 GB."
        }
        else {

            Write-ITLog `
                "CANH BAO" `
                "LENH GUI CANH BAO DUNG LUONG O D: <20 GB KHONG THANH CONG."
        }


        # -------------------------------------------------------------
        # Chờ 30 phút
        # -------------------------------------------------------------

        Write-ITLog `
            "CHO" `
            "CHO $WaitMinutes PHUT TRUOC KHI KIEM TRA LAI."


        Start-Sleep -Seconds $WaitSeconds


        # =================================================================
        # Kiểm tra lại sau 30 phút
        # =================================================================

        $VolumeAfterWait = Get-ITVolume `
            -GUID $VolumeGUID


        if ($null -eq $VolumeAfterWait) {

            throw "KHONG TIM THAY VOLUME KHI KIEM TRA LAI SAU $WaitMinutes PHUT."
        }


        $NewInfo = Get-ITDiskInfo `
            -Volume $VolumeAfterWait


        if ($null -eq $NewInfo) {

            throw "KHONG DOC DUOC DUNG LUONG SAU $WaitMinutes PHUT."
        }


        $NewFreeGB      = $NewInfo.FreeGB
    

        Write-ITLog `
            "KIEM TRA LAI" `
            "SAU $WaitMinutes PHUT: CON TRONG=$NewFreeGB GB."


        # =================================================================
        # Sau 30 phút nếu >=20 GB thì không ẩn D:
        # =================================================================

        if ($NewFreeGB -ge $HideGB) {

            Write-ITLog `
                "PHUC HOI" `
                "DUNG LUONG DA >=20 GB. KHONG AN O D:."


            $Message = @"
THONG BAO IT-MAINTENANCE

DUNG LUONG O D:
DA TRO LAI >=20 GB.

DUNG LUONG TRONG HIEN TAI:
$NewFreeGB GB

HE THONG SE KHONG AN O D:.

NGUOI DUNG CO THE TIEP TUC SU DUNG.

QUAN TRI VIEN
"@


            [void](Show-UserMessage -Message $Message)

            exit 0
        }


        # =================================================================
        # Sau 30 phút vẫn <20 GB: chuẩn bị gỡ ký tự D:
        # =================================================================

        Write-ITLog `
            "NGHIEM TRONG" `
            "SAU $WaitMinutes PHUT DUNG LUONG VAN <20 GB. BAT DAU AN O D:."


        # -------------------------------------------------------------
        # Xác nhận D: vẫn đang tồn tại
        # -------------------------------------------------------------

        $CurrentDrive = Get-CimInstance `
            Win32_Volume `
            -Filter "DriveLetter='D:'"


        if ($null -eq $CurrentDrive) {

            Write-ITLog `
                "THONG TIN" `
                "D: KHONG CON DUOC GAN. KHONG CAN REMOVE LAI."

            exit 0
        }


        # -------------------------------------------------------------
        # Bảo vệ: D: phải đúng Volume GUID đã cấu hình
        # -------------------------------------------------------------

        if ($CurrentDrive.DeviceID -ne $VolumeGUID) {

            throw "D: HIEN TAI KHONG PHAI VOLUME DA DUOC IT-MAINTENANCE CAU HINH. HUY THAO TAC DISKPART."
        }


        # -------------------------------------------------------------
        # Xác nhận Disk/Partition vẫn tồn tại
        # -------------------------------------------------------------

        $StoredPartition = Get-Partition `
            -DiskNumber $DiskNumber `
            -PartitionNumber $PartitionNumber `
            -ErrorAction SilentlyContinue


        if ($null -eq $StoredPartition) {

            throw "KHONG TIM THAY DISK/PARTITION DA LUU. HUY THAO TAC GO D:."
        }


        # -------------------------------------------------------------
        # DiskPart gỡ ký tự D:
        # -------------------------------------------------------------

        Write-ITLog `
            "THAO TAC" `
            "BAT DAU DISKPART REMOVE LETTER=D | DISK=$DiskNumber | PARTITION=$PartitionNumber"


        $DiskPartResult = Invoke-ITDiskPart @(
            "select disk $DiskNumber",
            "select partition $PartitionNumber",
            "remove letter=D",
            "exit"
        )


        Start-Sleep -Seconds 2


        # -------------------------------------------------------------
        # Kiểm tra lại bằng Volume GUID
        # -------------------------------------------------------------

        $CheckVolume = Get-ITVolume `
            -GUID $VolumeGUID


        if (
            $null -ne $CheckVolume -and
            $CheckVolume.DriveLetter -ne "D:"
        ) {

            Write-ITLog `
                "THANH CONG" `
                "DA GO KY TU O DIA D:."


            $Message = @"
THONG BAO IT-MAINTENANCE

O DIA D: DA DUOC TAM THOI AN.

LY DO:

SAU 60 PHUT,
DUNG LUONG TRONG VAN <20 GB.

DU LIEU HIEN CO KHONG BI XOA.

IT-MAINTENANCE VAN TIEP TUC
THEO DOI DUNG LUONG VOLUME.

KHI DUNG LUONG TRONG TRO LAI >=20 GB,
HE THONG SE TU DONG GAN LAI O D:.

QUAN TRI VIEN
"@


            [void](Show-UserMessage -Message $Message)


            Write-ITLog `
                "THONG BAO" `
                "DA GUI THONG BAO O D: DA DUOC TAM THOI AN."
        }
        else {

            Write-ITLog `
                "LOI" `
                "DISKPART KHONG GO DUOC D:. KET QUA=$DiskPartResult"


            $Message = @"
THONG BAO IT-MAINTENANCE

DUNG LUONG O D: VAN <20 GB.

HE THONG DA THU TAM THOI AN O D:
NHUNG THAO TAC KHONG THANH CONG.

VUI LONG LIEN HE QUAN TRI VIEN.

QUAN TRI VIEN
"@


            [void](Show-UserMessage -Message $Message)
        }


        exit 0
    }

}
catch {

    Write-ITLog `
        "LOI" `
        $_.Exception.Message

    exit 1
}
